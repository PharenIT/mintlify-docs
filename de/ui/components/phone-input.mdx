```vue
import { PhoneInput } from '@pharen/ui';
</script>

<template>
  <PhoneInput />
</template>
```

</Accordion>

## Installation

<CodeGroup>
```bash npm
npm install @pharen/ui
```

```bash pnpm
pnpm add @pharen/ui
```

```bash yarn
yarn add @pharen/ui
```

```bash bun
bun add @pharen/ui
```
</CodeGroup>

<Note>
  Importiere `@pharen/ui/styles.css` genau einmal im Einstiegspunkt deiner Anwendung.
</Note>

## Verwendung

```ts
import { PhoneInput } from '@pharen/ui';
```
