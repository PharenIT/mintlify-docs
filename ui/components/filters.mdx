Use **Filters** to build composable filtering controls. This page covers every Vue component exported by the public module.

<Note>
  Install the package and import its styles once. See [Installation](/ui/installation) for the complete setup.
</Note>

## Import

```ts
import { Filters, FilterInput, FilterValueSelector } from '@pharen/ui';
```

## Example

```vue
<script setup lang="ts">
import { Filters } from '@pharen/ui';
import { ref } from 'vue';
import type { Filter, FilterFieldConfig } from '@pharen/ui';

const filters = ref<Filter[]>([]);
const fields: FilterFieldConfig[] = [
  { id: 'status', label: 'Status', type: 'select', options: [
    { label: 'Active', value: 'active' },
    { label: 'Paused', value: 'paused' },
  ] },
];
</script>

<template>
  <Filters v-model="filters" :fields="fields" />
</template>
```
