```vue
<script setup lang="ts">
import { Context, ContextContent, ContextContentBody, ContextContentFooter, ContextContentHeader, ContextInputUsage, ContextOutputUsage, ContextTrigger } from '@pharen/ui/components/ai';
</script>

<template>
  <Context
    :used-tokens="18420"
    :max-tokens="128000"
    :default-open="true"
    model-id="GPT-5.4"
    :usage="{ inputTokens: 14200, outputTokens: 4220, costEur: 0.084, currency: 'EUR' }"
  >
    <ContextTrigger />
    <ContextContent>
      <ContextContentHeader />
      <ContextContentBody class="space-y-2"><ContextInputUsage /><ContextOutputUsage /></ContextContentBody>
      <ContextContentFooter />
    </ContextContent>
  </Context>
</template>
```

## Installation

```bash npm
npm install @pharen/ui
```

```bash pnpm
pnpm add @pharen/ui
```

```bash yarn
yarn add @pharen/ui
```

```bash bun
bun add @pharen/ui
```

</CodeGroup>

<Note>
  Import `@pharen/ui/styles.css` exactly once in your application entrypoint.
</Note>

## Usage

```ts
import { Context, ContextTrigger, ContextContent, ContextContentHeader, ContextContentBody, ContextContentFooter, ContextInputUsage, ContextOutputUsage } from '@pharen/ui/components/ai';
```

## Exports

| Export | Purpose |
| --- | --- |
| `Context` | Exported Vue component from this module. |
| `ContextTrigger` | Opens or activates the component. |
| `ContextContent` | Contains the visible content. |
| `ContextContentHeader` | Structures the header region. |
| `ContextContentBody` | Exported Vue component from this module. |
| `ContextContentFooter` | Structures the footer region. |
| `ContextInputUsage` | Exported Vue component from this module. |
| `ContextOutputUsage` | Exported Vue component from this module. |
